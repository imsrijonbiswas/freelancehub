Database: freelancehub

CREATE DATABASE IF NOT EXISTS freelancehub;
USE freelancehub;


CREATE TABLE skill_categories (
    skill_id INT AUTO_INCREMENT PRIMARY KEY,
    skill_name VARCHAR(100) NOT NULL UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE freelancers (
    freelancer_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    skill_id INT NULL,
    is_verified BOOLEAN DEFAULT FALSE,
    is_suspended BOOLEAN DEFAULT FALSE,
    bio TEXT,
    profile_picture VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_freelancers_skill
        FOREIGN KEY (skill_id)
        REFERENCES skill_categories(skill_id)
        ON DELETE SET NULL
);


CREATE TABLE clients (
    client_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    is_suspended BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE admins (
    admin_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100),
    email VARCHAR(150) UNIQUE,
    password VARCHAR(255)
);


CREATE TABLE questions (
    question_id INT AUTO_INCREMENT PRIMARY KEY,
    skill_id INT NOT NULL,
    question_text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_question_skill
        FOREIGN KEY (skill_id)
        REFERENCES skill_categories(skill_id)
        ON DELETE CASCADE
);


CREATE TABLE question_options (
    option_id INT AUTO_INCREMENT PRIMARY KEY,
    question_id INT NOT NULL,
    option_label CHAR(1) NOT NULL,
    option_text TEXT NOT NULL,
    is_correct BOOLEAN DEFAULT FALSE,

    CONSTRAINT fk_option_question
        FOREIGN KEY (question_id)
        REFERENCES questions(question_id)
        ON DELETE CASCADE,

    CONSTRAINT chk_option_label
        CHECK (option_label IN ('A','B','C','D'))
);


CREATE TABLE skill_tests (
    test_id INT AUTO_INCREMENT PRIMARY KEY,
    skill_id INT NOT NULL,
    total_questions INT DEFAULT 15,
    pass_percentage DECIMAL(5,2) DEFAULT 70.00,

    FOREIGN KEY (skill_id) REFERENCES skill_categories(skill_id)
);


CREATE TABLE test_attempts (
    attempt_id INT AUTO_INCREMENT PRIMARY KEY,
    freelancer_id INT NOT NULL,
    test_id INT NOT NULL,
    score DECIMAL(5,2),
    passed BOOLEAN,
    attempted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (freelancer_id) REFERENCES freelancers(freelancer_id),
    FOREIGN KEY (test_id) REFERENCES skill_tests(test_id)
);


CREATE TABLE test_answers (
    answer_id INT AUTO_INCREMENT PRIMARY KEY,
    attempt_id INT NOT NULL,
    question_id INT NOT NULL,
    selected_option CHAR(1),

    FOREIGN KEY (attempt_id) REFERENCES test_attempts(attempt_id),
    FOREIGN KEY (question_id) REFERENCES questions(question_id)
);


CREATE TABLE jobs (
    job_id INT AUTO_INCREMENT PRIMARY KEY,
    client_id INT NOT NULL,
    skill_id INT NOT NULL,
    job_title VARCHAR(150),
    job_description TEXT,
    status ENUM('open', 'closed') DEFAULT 'open',
    is_hired BOOLEAN DEFAULT FALSE,
    budget DECIMAL(10,2),
    hired_freelancer_id INT DEFAULT NULL,
    hired_at TIMESTAMP NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (client_id) REFERENCES clients(client_id),
    FOREIGN KEY (skill_id) REFERENCES skill_categories(skill_id),
    FOREIGN KEY (hired_freelancer_id)
        REFERENCES freelancers(freelancer_id)
        ON DELETE SET NULL
);


CREATE TABLE job_applications (
    application_id INT AUTO_INCREMENT PRIMARY KEY,
    job_id INT NOT NULL,
    freelancer_id INT NOT NULL,
    applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (job_id) REFERENCES jobs(job_id),
    FOREIGN KEY (freelancer_id) REFERENCES freelancers(freelancer_id)
);


CREATE TABLE messages (
    message_id INT AUTO_INCREMENT PRIMARY KEY,
    job_id INT NOT NULL,
    sender_type ENUM('freelancer', 'client') NOT NULL,
    sender_id INT NOT NULL,
    message_text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (job_id) REFERENCES jobs(job_id) ON DELETE CASCADE
);


CREATE TABLE hired_freelancers (
    hire_id INT AUTO_INCREMENT PRIMARY KEY,
    job_id INT NOT NULL UNIQUE,
    freelancer_id INT NOT NULL,
    client_id INT NOT NULL,
    hired_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status ENUM('active', 'completed', 'cancelled') DEFAULT 'active',

    FOREIGN KEY (job_id) REFERENCES jobs(job_id) ON DELETE CASCADE,
    FOREIGN KEY (freelancer_id) REFERENCES freelancers(freelancer_id) ON DELETE CASCADE,
    FOREIGN KEY (client_id) REFERENCES clients(client_id) ON DELETE CASCADE
);


CREATE TABLE ratings (
    rating_id INT AUTO_INCREMENT PRIMARY KEY,
    hire_id INT NOT NULL,
    rating INT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    review TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (hire_id) REFERENCES hired_freelancers(hire_id) ON DELETE CASCADE
);


CREATE TABLE payments (
    payment_id INT AUTO_INCREMENT PRIMARY KEY,
    hire_id INT NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    payment_status ENUM('pending', 'completed', 'failed') DEFAULT 'pending',
    paid_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (hire_id) REFERENCES hired_freelancers(hire_id) ON DELETE CASCADE
);



CREATE TABLE courses (
    course_id INT AUTO_INCREMENT PRIMARY KEY,
    course_title VARCHAR(200),
    course_url TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE skill_courses (
    skill_id INT NOT NULL,
    course_id INT NOT NULL,

    PRIMARY KEY (skill_id, course_id),

    FOREIGN KEY (skill_id) REFERENCES skill_categories(skill_id),
    FOREIGN KEY (course_id) REFERENCES courses(course_id)
);


CREATE TABLE user_reports (
    report_id INT AUTO_INCREMENT PRIMARY KEY,
    reported_user_id INT NOT NULL,
    reported_user_type ENUM('freelancer', 'client') NOT NULL,
    reporter_id INT,
    reporter_type ENUM('freelancer', 'client', 'admin'),
    job_id INT NULL,
    report_reason TEXT NOT NULL,
    report_message TEXT,
    status ENUM('pending', 'reviewed', 'resolved') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    UNIQUE KEY unique_job_report (job_id, reported_user_id, reporter_id)
);


CREATE TABLE freelancer_test_attempts (
    attempt_id INT AUTO_INCREMENT PRIMARY KEY,
    freelancer_email VARCHAR(150) NOT NULL,
    skill_id INT NOT NULL,
    score DECIMAL(5,2) NOT NULL,
    passed BOOLEAN NOT NULL,
    attempted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (skill_id) REFERENCES skill_categories(skill_id) ON DELETE CASCADE
);
